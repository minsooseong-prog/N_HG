/************************************************************* * 헬스 기구 챗봇 - Apps Script 백엔드 (Code.gs) * * 흐름: * 1) 구글 드라이브에 올린 운동 데이터셋(JSON)을 파일 ID로 읽어옴 * 2) 사용자가 찍은 기구 사진을 Gemini Vision에 전달 * 3) Gemini가 기구를 식별하고 설명/사용법/타깃부위/주의점을 생성 * 4) 드라이브 데이터셋에서 해당 기구 관련 운동을 찾아 보강 * 5) 사용자가 후속 질문을 하면 식별된 기구를 맥락으로 답변 *************************************************************/ // ===== 설정 ===================================================== // Gemini API 키는 코드에 직접 쓰지 말고 스크립트 속성에 저장하세요. // 확장 프로그램 > Apps Script > 프로젝트 설정(톱니) > 스크립트 속성 // 속성 이름: GEMINI_API_KEY 값: 본인 키 const GEMINI_API_KEY = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY'); // 구글 드라이브에 업로드한 데이터셋 JSON 파일의 ID (공유 링크 가운데 부분) const DATASET_FILE_ID = '1T9USw4ZfIj4a9N2FUDJapwoldM6GmIKH'; // 사용할 Gemini 모델. gemini-2.0-flash 는 2026-06-01 종료됨. // 최신: gemini-3.5-flash / 또는 gemini-2.5-flash 사용 가능. 본인 모델로 맞추세요. const GEMINI_MODEL = 'gemini-2.5-flash'; const GEMINI_ENDPOINT = 'https://generativelanguage.googleapis.com/v1beta/models/' + GEMINI_MODEL + ':generateContent?key=' + GEMINI_API_KEY; // ===== 웹앱 진입점 ============================================== function doGet() { return HtmlService.createHtmlOutputFromFile('Sidebar') .setTitle('헬스 기구 챗봇') .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } // ===== 드라이브 데이터셋 로드 ==================================== /** * 드라이브의 JSON 데이터셋을 읽어 배열로 반환합니다. * 매 요청마다 파일을 다시 읽지 않도록 6시간 캐싱합니다. * (CacheService는 값당 100KB 제한이 있어, 큰 파일은 캐시를 건너뛰고 * 매번 드라이브에서 읽습니다. 그래도 동작에는 문제 없습니다.) */ // 같은 실행(요청) 안에서 드라이브를 여러 번 읽지 않도록 메모리에 보관 var _datasetMemo = null; function loadDataset() { if (_datasetMemo) { return _datasetMemo; } const cache = CacheService.getScriptCache(); const cached = cache.get('exerciseDataset'); if (cached) { _datasetMemo = JSON.parse(cached); return _datasetMemo; } const file = DriveApp.getFileById(DATASET_FILE_ID); const content = file.getBlob().getDataAsString('UTF-8'); const data = JSON.parse(content); // 데이터셋이 { "exercises": [...] } 처럼 감싸져 있을 수도 있으니 처리 const list = Array.isArray(data) ? data : (data.exercises || data.data || []); try { cache.put('exerciseDataset', JSON.stringify(list), 21600); // 6시간 } catch (e) { // 100KB 초과 시 캐시 생략 (정상 동작) } _datasetMemo = list; return list; } /** * [디버그용] 데이터셋의 첫 항목 구조를 로그로 확인합니다. * 실행 후 [실행 로그]에서 실제 필드명(name, equipment, target 등)을 확인하고, * 아래 normalizeEntry / getEquipmentTypes 의 필드명을 맞추세요. */ function inspectDataset() { const list = loadDataset(); Logger.log('총 항목 수: ' + list.length); Logger.log('첫 항목: ' + JSON.stringify(list[0], null, 2)); } /** * [1회 실행 유틸] 큰 원본 데이터셋을 경량 JSON으로 변환해 드라이브에 새로 저장합니다. * - 다국어 설명(it, tr)과 instructions/gif/image 등 무거운 필드를 제거 * - "관련 운동" 기능에 필요한 핵심 필드만 유지 → 파일 크기 대폭 감소 * * 실행 후 [실행 로그]에 찍히는 새 파일 ID를 복사해 * 상단 DATASET_FILE_ID 값을 그 ID로 바꾸면 됩니다. (한 번만 하면 됨) */ function buildSlimDataset() { const list = loadDataset(); const slim = list.map(function (e) { return { id: e.id, name: e.name, equipment: e.equipment, target: e.target, muscle_group: e.muscle_group, secondary_muscles: e.secondary_muscles || [], body_part: e.body_part, category: e.category }; }); const content = JSON.stringify(slim); const file = DriveApp.createFile('exercises_slim.json', content, 'application/json'); Logger.log('경량 파일 생성 완료'); Logger.log('파일 크기: ' + Math.round(content.length / 1024) + ' KB'); Logger.log('새 파일 ID (이 값을 DATASET_FILE_ID 에 넣으세요): ' + file.getId()); } /** * 데이터셋의 항목을 공통 형식으로 정리합니다. * 데이터셋마다 필드명이 다를 수 있어 여러 후보를 시도합니다. * inspectDataset() 결과를 보고 필요하면 필드명을 수정하세요. */ function normalizeEntry(e) { return { name: e.name || '', equipment: e.equipment || '', target: e.target || e.muscle_group || '', secondary: e.secondary_muscles || [], // instructions 는 {en, it, tr} 다국어 객체이므로 단계 배열은 instruction_steps.en 사용 instructions: (e.instruction_steps && e.instruction_steps.en) || [] }; } /** * 데이터셋에 존재하는 고유 equipment(장비 종류) 목록을 추출합니다. * Gemini에게 "이 목록 중 가장 맞는 값을 골라" 라고 시켜 매칭 정확도를 높입니다. */ function getEquipmentTypes() { const list = loadDataset(); const set = {}; list.forEach(function (e) { const eq = normalizeEntry(e).equipment; if (eq) set[String(eq).toLowerCase().trim()] = true; }); return Object.keys(set); } // ===== 메인: 사진 분석 ========================================== /** * 사이드바에서 호출. 사진을 분석해 기구 정보를 반환합니다. * @param {string} base64Image - data: 접두어 없는 순수 base64 문자열 * @param {string} mimeType - 예: 'image/jpeg', 'image/png' */ function analyzeEquipment(base64Image, mimeType) { if (!GEMINI_API_KEY) { throw new Error('GEMINI_API_KEY가 설정되지 않았습니다. 스크립트 속성을 확인하세요.'); } const equipmentTypes = getEquipmentTypes(); // 그라운딩용 장비 목록 (보통 수십 개) const prompt = '당신은 헬스 트레이닝 전문가입니다. 첨부된 사진 속 헬스 기구를 식별하고, ' + '한국 헬스장 이용자를 위해 한국어로 정보를 정리하세요.\n\n' + '아래 "장비 분류 목록"은 데이터셋에 존재하는 영어 장비 종류입니다. ' + 'matchedEquipment 필드에는 이 목록 중 사진과 가장 잘 맞는 값 하나를 그대로 적으세요. ' + '적절한 값이 없으면 빈 문자열로 두세요.\n' + '장비 분류 목록: ' + JSON.stringify(equipmentTypes) + '\n\n' + '반드시 아래 JSON 형식 하나만 출력하세요 (다른 텍스트, 마크다운 금지):\n' + '{\n' + ' "equipmentNameKo": "기구의 한국어 이름",\n' + ' "equipmentNameEn": "기구의 영어 이름",\n' + ' "matchedEquipment": "장비 분류 목록 중 택1 또는 빈 문자열",\n' + ' "description": "기구에 대한 2~3문장 설명",\n' + ' "usage": ["사용 방법을 단계별로 3~6개"],\n' + ' "targetMuscles": ["주로 자극되는 부위들"],\n' + ' "cautions": ["부상 예방 등 주의할 점 2~4개"],\n' + ' "confidence": "high | medium | low (식별 확신도)"\n' + '}\n' + '사진에 헬스 기구가 없거나 식별이 어려우면 confidence를 low로 하고 ' + 'description에 그 사실을 적으세요.'; const payload = { contents: [{ parts: [ { text: prompt }, { inline_data: { mime_type: mimeType, data: base64Image } } ] }], generationConfig: { responseMimeType: 'application/json' } }; const raw = callGemini(payload); const result = safeParseJson(raw); // 드라이브 데이터셋에서 관련 운동 보강 result.relatedExercises = findRelatedExercises(result.matchedEquipment); return result; } /** * matchedEquipment 값과 일치하는 운동을 데이터셋에서 찾아 최대 6개 반환합니다. */ function findRelatedExercises(matchedEquipment) { if (!matchedEquipment) return []; const target = String(matchedEquipment).toLowerCase().trim(); const list = loadDataset(); const out = []; for (let i = 0; i < list.length && out.length < 6; i++) { const e = normalizeEntry(list[i]); if (String(e.equipment).toLowerCase().trim() === target) { out.push({ name: e.name, target: e.target }); } } return out; } // ===== 후속 질문 챗봇 =========================================== /** * 식별된 기구를 맥락으로 후속 질문에 답합니다. * @param {string} question - 사용자 질문 * @param {Object} equipmentInfo - analyzeEquipment 가 반환한 결과 객체 * @param {Array} history - [{role:'user'|'model', text:'...'}] 형태의 대화 이력 */ function askFollowUp(question, equipmentInfo, history) { if (!GEMINI_API_KEY) { throw new Error('GEMINI_API_KEY가 설정되지 않았습니다.'); } const systemContext = '당신은 친절한 헬스 트레이닝 코치입니다. 사용자가 다음 기구에 대해 질문합니다. ' + '한국어로 간결하고 정확하게 답하세요. 자세나 안전이 걱정되는 부분은 ' + '"정확한 자세는 전문 트레이너에게 확인하세요"라고 안내하세요.\n\n' + '대상 기구 정보:\n' + JSON.stringify(equipmentInfo, null, 2); const contents = []; // 첫 메시지에 맥락을 넣어줌 (Gemini REST에는 별도 system role 대신 첫 user 턴 활용) contents.push({ role: 'user', parts: [{ text: systemContext }] }); contents.push({ role: 'model', parts: [{ text: '네, 이 기구에 대해 무엇이든 물어보세요.' }] }); (history || []).forEach(function (h) { contents.push({ role: h.role, parts: [{ text: h.text }] }); }); contents.push({ role: 'user', parts: [{ text: question }] }); const payload = { contents: contents }; return callGemini(payload); } // ===== 공통 유틸 ================================================ function callGemini(payload) { const options = { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true }; const resp = UrlFetchApp.fetch(GEMINI_ENDPOINT, options); const code = resp.getResponseCode(); const text = resp.getContentText(); if (code !== 200) { throw new Error('Gemini API 오류 (' + code + '): ' + text); } const data = JSON.parse(text); if (!data.candidates || !data.candidates.length) { throw new Error('Gemini 응답에 결과가 없습니다: ' + text); } const parts = data.candidates[0].content.parts || []; return parts.map(function (p) { return p.text || ''; }).join(''); } /** * Gemini가 가끔 ```json 펜스를 붙이는 경우까지 대비해 안전하게 파싱합니다. */ function safeParseJson(raw) { let s = String(raw).trim(); s = s.replace(/^```json\s*/i, '').replace(/^```\s*/, '').replace(/```\s*$/, ''); try { return JSON.parse(s); } catch (e) { // 본문 중 첫 { ... } 블록만 추출 재시도 const m = s.match(/\{[\s\S]*\}/); if (m) return JSON.parse(m[0]); throw new Error('JSON 파싱 실패: ' + raw); } }
